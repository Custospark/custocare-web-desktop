import React, { useState, useEffect, useCallback, useMemo } from 'react';
import { useNavigate, Link } from 'react-router-dom';
import { 
  Building2, 
  Stethoscope, 
  User, 
  ChevronRight,
  LogOut,
  Quote,
  Shield,
  Activity,
  Moon,
  Sun,
  Check,
  Sparkles,
  HelpCircle,
  ArrowRight,
  Lock,
  Users
} from 'lucide-react';
import { useAppDispatch, useAppSelector } from '../../../../app/store/hooks/useApp';
import { toggleTheme } from '../../../../app/store/slices/uiSlice';
import { cn } from '../../../../shared/types/cn';
import { motion, AnimatePresence } from 'framer-motion';

interface RoleOption {
  id: string;
  title: string;
  description: string;
  icon: React.ReactNode;
  gradient: string;
  features: string[];
  badge?: string;
}

const RoleSelectionScreen: React.FC = () => {
  const dispatch = useAppDispatch();
  const navigate = useNavigate();
  const theme = useAppSelector((state) => state.ui.theme);
  const [selectedRole, setSelectedRole] = useState<string | null>(null);
  const [isTransitioning, setIsTransitioning] = useState(false);
  const [hoveredRole, setHoveredRole] = useState<string | null>(null);
  const [hasInteracted, setHasInteracted] = useState(false);

  // Role options with enhanced data
  const roleOptions: RoleOption[] = useMemo(() => [
    {
      id: 'facility-owner',
      title: 'Facility Owner',
      description: 'For administrators managing clinics, hospitals, or care facilities.',
      icon: <Building2 className="w-5 h-5" />,
      gradient: 'from-emerald-500 to-teal-600',
      features: ['Multi-facility Dashboard', 'Staff Management', 'Billing & Reporting', 'Compliance Tools'],
      badge: 'Enterprise'
    },
    {
      id: 'medical-professional',
      title: 'Medical Professional',
      description: 'For doctors, nurses, and support staff providing direct care.',
      icon: <Stethoscope className="w-5 h-5" />,
      gradient: 'from-blue-500 to-indigo-600',
      features: ['Patient Records', 'Clinical Tools', 'AI-Assisted Diagnostics', 'Team Collaboration'],
      badge: 'Clinical'
    },
    {
      id: 'patient',
      title: 'Patient',
      description: 'For individuals seeking care, monitoring health, or managing appointments.',
      icon: <User className="w-5 h-5" />,
      gradient: 'from-purple-500 to-pink-600',
      features: ['Appointment Booking', 'Health Tracking', 'Secure Messaging', 'Prescription Management']
    }
  ], []);

  // Images with fallbacks
  const images = useMemo(() => ({
    testimonial: 'https://images.unsplash.com/photo-1612349317150-e413f6a5b16d?w=800&h=800&fit=crop&q=80',
    background: 'https://images.unsplash.com/photo-1576091160399-112ba8d25d1d?w=1600&h=900&fit=crop&q=80',
    fallbackAvatar: 'https://images.unsplash.com/photo-1494790108755-2616b612b786?w=400&h=400&fit=crop&q=80'
  }), []);

  // Animation variants
  const containerVariants = {
    hidden: { opacity: 0 },
    visible: {
      opacity: 1,
      transition: {
        staggerChildren: 0.1,
        delayChildren: 0.2
      }
    }
  };


  // Handle role selection with feedback
  const handleRoleSelect = useCallback((roleId: string) => {
    if (!hasInteracted) setHasInteracted(true);
    
    if (selectedRole === roleId) {
      setSelectedRole(null);
    } else {
      setSelectedRole(roleId);
      
      // Add haptic feedback if available
      if (window.navigator.vibrate) {
        window.navigator.vibrate(50);
      }
    }
  }, [selectedRole, hasInteracted]);

  // Handle continue with smooth transition
  const handleContinue = useCallback(() => {
    if (!selectedRole) return;
    
    setIsTransitioning(true);
    
    // Simulate API call or validation
    setTimeout(() => {
      // Analytics tracking
      console.log(`User selected role: ${selectedRole}`);
      
      // Navigate to role-specific onboarding
      navigate(`/onboarding/${selectedRole}`, {
        state: { role: selectedRole }
      });
    }, 600);
  }, [selectedRole, navigate]);

  // Keyboard navigation support
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape') {
        setSelectedRole(null);
      }
      
      if (e.key === 'Enter' && selectedRole) {
        handleContinue();
      }
      
      // Arrow key navigation between roles
      if (roleOptions.length > 0) {
        const currentIndex = roleOptions.findIndex(role => role.id === selectedRole);
        
        if (e.key === 'ArrowDown' || e.key === 'ArrowRight') {
          e.preventDefault();
          const nextIndex = (currentIndex + 1) % roleOptions.length;
          setSelectedRole(roleOptions[nextIndex].id);
        }
        
        if (e.key === 'ArrowUp' || e.key === 'ArrowLeft') {
          e.preventDefault();
          const prevIndex = currentIndex <= 0 ? roleOptions.length - 1 : currentIndex - 1;
          setSelectedRole(roleOptions[prevIndex].id);
        }
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [selectedRole, roleOptions, handleContinue]);

  // Get selected role details
  const selectedRoleDetails = useMemo(() => 
    roleOptions.find(role => role.id === selectedRole),
    [selectedRole, roleOptions]
  );

  return (
    <motion.div 
      initial={{ opacity: 0 }}
      animate={{ opacity: 1 }}
      exit={{ opacity: 0 }}
      className={cn(
        'min-h-screen flex flex-col lg:flex-row overflow-hidden',
        theme === 'dark'
          ? 'bg-gradient-to-br from-gray-950 via-gray-900 to-gray-950'
          : 'bg-gradient-to-br from-slate-50 via-white to-slate-100'
      )}
    >
      {/* Left Panel - Main Content */}
      <div className="flex-1 flex flex-col relative">
        {/* Animated background elements */}
        <div className="absolute inset-0 overflow-hidden pointer-events-none">
          <div className={cn(
            'absolute top-0 right-0 w-64 h-64 rounded-full blur-3xl opacity-10',
            theme === 'dark' ? 'bg-cyan-500' : 'bg-blue-300'
          )} />
          <div className={cn(
            'absolute bottom-0 left-0 w-96 h-96 rounded-full blur-3xl opacity-10',
            theme === 'dark' ? 'bg-purple-500' : 'bg-purple-300'
          )} />
        </div>

        {/* Header */}
        <motion.header 
          initial={{ y: -20, opacity: 0 }}
          animate={{ y: 0, opacity: 1 }}
          transition={{ duration: 0.3 }}
          className="relative z-10 flex items-center justify-between px-5 py-4 lg:px-8 lg:py-6"
        >
          <Link 
            to="/" 
            className="inline-flex items-center gap-2.5 group relative"
            aria-label="Go to homepage"
          >
            <motion.div 
              whileHover={{ scale: 1.05, rotate: 5 }}
              whileTap={{ scale: 0.95 }}
              className={cn(
                'w-10 h-10 rounded-xl flex items-center justify-center shadow-lg',
                'relative overflow-hidden',
                theme === 'dark'
                  ? 'bg-gradient-to-br from-cyan-500 to-blue-600'
                  : 'bg-gradient-to-br from-blue-500 to-cyan-600'
              )}
            >
              <Activity className="w-5 h-5 text-white relative z-10" strokeWidth={2.5} />
              <div className="absolute inset-0 bg-white/10 backdrop-blur-sm" />
            </motion.div>
            
            <div className="flex flex-col">
              <span className={cn(
                'text-lg font-bold leading-tight tracking-tight',
                theme === 'dark' ? 'text-white' : 'text-gray-900'
              )}>
                CustoCare AI
              </span>
              <span className={cn(
                'text-xs tracking-wide',
                theme === 'dark' ? 'text-gray-400' : 'text-gray-500'
              )}>
                Medical Decision Support
              </span>
            </div>
          </Link>

          <div className="flex items-center gap-2">
            <motion.button
              whileHover={{ scale: 1.05 }}
              whileTap={{ scale: 0.95 }}
              onClick={() => dispatch(toggleTheme())}
              className={cn(
                'p-2.5 rounded-xl transition-all duration-200',
                'focus:outline-none focus:ring-2 focus:ring-offset-2',
                theme === 'dark'
                  ? 'bg-gray-800/80 hover:bg-gray-700/80 text-amber-300 focus:ring-cyan-500/50 focus:ring-offset-gray-900'
                  : 'bg-white/80 hover:bg-white text-gray-600 focus:ring-blue-500/50 focus:ring-offset-white backdrop-blur-sm shadow-sm'
              )}
              aria-label={`Switch to ${theme === 'dark' ? 'light' : 'dark'} mode`}
            >
              {theme === 'dark' ? (
                <Sun className="w-4 h-4" />
              ) : (
                <Moon className="w-4 h-4" />
              )}
            </motion.button>

            <motion.button
              whileHover={{ scale: 1.05 }}
              whileTap={{ scale: 0.95 }}
              className={cn(
                'flex items-center gap-2 px-3.5 py-2 rounded-xl text-sm font-medium',
                'transition-all focus:outline-none focus:ring-2 focus:ring-offset-2',
                theme === 'dark'
                  ? 'bg-gray-800/80 text-gray-300 hover:bg-gray-700/80 hover:text-white focus:ring-cyan-500/50 focus:ring-offset-gray-900'
                  : 'bg-white/80 text-gray-600 hover:bg-white hover:text-gray-900 focus:ring-blue-500/50 focus:ring-offset-white backdrop-blur-sm shadow-sm'
              )}
              aria-label="Log out"
            >
              <LogOut className="w-4 h-4" />
              <span className="hidden sm:inline">Log Out</span>
            </motion.button>
          </div>
        </motion.header>

        {/* Main Content */}
        <main className="flex-1 flex items-center justify-center px-5 py-4 lg:px-8 lg:py-6 overflow-y-auto relative z-10">
          <div className="w-full max-w-3xl">
            {/* Header Section */}
            <motion.div 
              initial={{ y: 20, opacity: 0 }}
              animate={{ y: 0, opacity: 1 }}
              transition={{ delay: 0.1 }}
              className="text-center mb-12"
            >
              <div className="inline-flex items-center gap-2 mb-4 px-3 py-1.5 rounded-full bg-gradient-to-r from-blue-500/10 to-cyan-500/10">
                <Sparkles className="w-4 h-4 text-blue-500" />
                <span className="text-sm font-medium text-blue-600 dark:text-cyan-400">
                  New Onboarding Experience
                </span>
              </div>
              
              <h1 className={cn(
                'text-3xl lg:text-4xl font-bold mb-4 tracking-tight',
                theme === 'dark' ? 'text-white' : 'text-gray-900'
              )}>
                Welcome to <span className="bg-gradient-to-r from-blue-600 to-cyan-500 bg-clip-text text-transparent">CustoCare AI</span>
              </h1>
              
              <h2 className={cn(
                'text-xl lg:text-2xl font-semibold mb-3',
                theme === 'dark' ? 'text-gray-300' : 'text-gray-700'
              )}>
                Tell us who you are
              </h2>
              
              <p className={cn(
                'max-w-2xl mx-auto text-base leading-relaxed',
                theme === 'dark' ? 'text-gray-400' : 'text-gray-600'
              )}>
                Select your primary role below so we can customize your onboarding experience. 
                Your choice will determine the tools and features you have access to.
              </p>
            </motion.div>

            {/* Role Selection Cards */}
            <motion.div 
              variants={containerVariants}
              initial="hidden"
              animate="visible"
              className="grid grid-cols-1 md:grid-cols-3 gap-4 mb-10"
            >
              {roleOptions.map((role) => (
                <motion.div
                  key={role.id}
                  whileHover="hover"
                  animate="rest"
                  onMouseEnter={() => setHoveredRole(role.id)}
                  onMouseLeave={() => setHoveredRole(null)}
                >
                  <button
                    onClick={() => handleRoleSelect(role.id)}
                    onKeyDown={(e) => {
                      if (e.key === 'Enter' || e.key === ' ') {
                        e.preventDefault();
                        handleRoleSelect(role.id);
                      }
                    }}
                    className={cn(
                      'w-full text-left p-6 rounded-2xl border transition-all duration-300',
                      'focus:outline-none focus:ring-2 focus:ring-offset-2 relative overflow-hidden',
                      'group',
                      selectedRole === role.id
                        ? theme === 'dark'
                          ? 'bg-gradient-to-br from-gray-800 to-gray-900 border-blue-500/50 ring-2 ring-blue-500/30'
                          : 'bg-gradient-to-br from-white to-gray-50 border-blue-500 ring-2 ring-blue-500/20 shadow-lg'
                        : theme === 'dark'
                          ? 'bg-gray-800/50 border-gray-700 hover:bg-gray-800/80 hover:border-gray-600 focus:ring-blue-500/30 focus:ring-offset-gray-900'
                          : 'bg-white border-gray-200 hover:bg-gray-50 hover:border-gray-300 focus:ring-blue-500/20 focus:ring-offset-white shadow-sm hover:shadow-md'
                    )}
                    aria-pressed={selectedRole === role.id}
                    aria-label={`Select ${role.title} role`}
                  >
                    {/* Selection indicator */}
                    <AnimatePresence>
                      {selectedRole === role.id && (
                        <motion.div
                          initial={{ scale: 0, opacity: 0 }}
                          animate={{ scale: 1, opacity: 1 }}
                          exit={{ scale: 0, opacity: 0 }}
                          className={cn(
                            'absolute top-3 right-3 w-6 h-6 rounded-full flex items-center justify-center',
                            theme === 'dark' ? 'bg-blue-500' : 'bg-blue-600'
                          )}
                        >
                          <Check className="w-3 h-3 text-white" />
                        </motion.div>
                      )}
                    </AnimatePresence>

                    {/* Role badge */}
                    {role.badge && (
                      <div className="absolute top-3 left-3">
                        <span className={cn(
                          'text-xs font-semibold px-2 py-1 rounded-full',
                          theme === 'dark' ? 'bg-blue-900/30 text-blue-300' : 'bg-blue-100 text-blue-700'
                        )}>
                          {role.badge}
                        </span>
                      </div>
                    )}

                    <div className="flex flex-col items-center text-center pt-2">
                      {/* Icon with gradient background */}
                      <div className={cn(
                        'w-16 h-16 rounded-2xl flex items-center justify-center mb-4',
                        'bg-gradient-to-br',
                        role.gradient,
                        selectedRole === role.id && 'shadow-lg'
                      )}>
                        <div className="text-white transform group-hover:scale-110 transition-transform duration-300">
                          {role.icon}
                        </div>
                      </div>

                      {/* Role title */}
                      <h3 className={cn(
                        'text-lg font-semibold mb-2',
                        selectedRole === role.id
                          ? theme === 'dark'
                            ? 'text-blue-300'
                            : 'text-blue-700'
                          : theme === 'dark'
                            ? 'text-white'
                            : 'text-gray-900'
                      )}>
                        {role.title}
                      </h3>

                      {/* Description */}
                      <p className={cn(
                        'text-sm mb-4 leading-relaxed',
                        selectedRole === role.id
                          ? theme === 'dark'
                            ? 'text-blue-100/80'
                            : 'text-blue-600/90'
                          : theme === 'dark'
                            ? 'text-gray-400'
                            : 'text-gray-600'
                      )}>
                        {role.description}
                      </p>

                      {/* Features list - only show on hover/select */}
                      <AnimatePresence>
                        {(hoveredRole === role.id || selectedRole === role.id) && (
                          <motion.div
                            initial={{ height: 0, opacity: 0 }}
                            animate={{ height: 'auto', opacity: 1 }}
                            exit={{ height: 0, opacity: 0 }}
                            className="w-full overflow-hidden"
                          >
                            <div className="pt-4 border-t border-gray-200 dark:border-gray-700">
                              <ul className="space-y-2 text-left">
                                {role.features.map((feature, index) => (
                                  <motion.li
                                    key={index}
                                    initial={{ x: -10, opacity: 0 }}
                                    animate={{ x: 0, opacity: 1 }}
                                    transition={{ delay: index * 0.1 }}
                                    className="flex items-center gap-2 text-xs"
                                  >
                                    <Check className={cn(
                                      'w-3 h-3 flex-shrink-0',
                                      theme === 'dark' ? 'text-blue-400' : 'text-blue-600'
                                    )} />
                                    <span className={theme === 'dark' ? 'text-gray-300' : 'text-gray-700'}>
                                      {feature}
                                    </span>
                                  </motion.li>
                                ))}
                              </ul>
                            </div>
                          </motion.div>
                        )}
                      </AnimatePresence>
                    </div>
                  </button>
                </motion.div>
              ))}
            </motion.div>

            {/* Selected Role Preview */}
            <AnimatePresence>
              {selectedRole && selectedRoleDetails && (
                <motion.div
                  initial={{ height: 0, opacity: 0 }}
                  animate={{ height: 'auto', opacity: 1 }}
                  exit={{ height: 0, opacity: 0 }}
                  className="mb-10 overflow-hidden"
                >
                  <div className={cn(
                    'rounded-2xl p-6 border',
                    theme === 'dark'
                      ? 'bg-gradient-to-r from-blue-900/20 to-cyan-900/20 border-blue-500/30'
                      : 'bg-gradient-to-r from-blue-50 to-cyan-50 border-blue-200'
                  )}>
                    <div className="flex items-center justify-between">
                      <div>
                        <div className="flex items-center gap-3 mb-2">
                          <div className={cn(
                            'w-10 h-10 rounded-xl flex items-center justify-center',
                            'bg-gradient-to-br',
                            selectedRoleDetails.gradient
                          )}>
                            {selectedRoleDetails.icon}
                          </div>
                          <div>
                            <h4 className={cn(
                              'text-lg font-semibold',
                              theme === 'dark' ? 'text-white' : 'text-gray-900'
                            )}>
                              {selectedRoleDetails.title}
                            </h4>
                            <p className={cn(
                              'text-sm',
                              theme === 'dark' ? 'text-gray-400' : 'text-gray-600'
                            )}>
                              Selected for onboarding
                            </p>
                          </div>
                        </div>
                        <p className={cn(
                          'text-sm',
                          theme === 'dark' ? 'text-gray-300' : 'text-gray-700'
                        )}>
                          You'll have access to: {selectedRoleDetails.features.join(', ')}
                        </p>
                      </div>
                      <ChevronRight className={cn(
                        'w-5 h-5',
                        theme === 'dark' ? 'text-blue-400' : 'text-blue-600'
                      )} />
                    </div>
                  </div>
                </motion.div>
              )}
            </AnimatePresence>

            {/* Continue Section */}
            <motion.div 
              initial={{ y: 20, opacity: 0 }}
              animate={{ y: 0, opacity: 1 }}
              transition={{ delay: 0.4 }}
              className="text-center mb-10"
            >
              <button
                onClick={handleContinue}
                disabled={!selectedRole || isTransitioning}
                className={cn(
                  'group relative inline-flex items-center justify-center',
                  'px-8 py-4 rounded-xl font-semibold text-base',
                  'transition-all duration-300 transform',
                  'focus:outline-none focus:ring-2 focus:ring-offset-2',
                  'disabled:cursor-not-allowed disabled:opacity-50',
                  selectedRole && !isTransitioning
                    ? theme === 'dark'
                      ? 'bg-gradient-to-r from-blue-600 to-cyan-500 hover:from-blue-500 hover:to-cyan-400 text-white shadow-lg hover:shadow-xl focus:ring-blue-500 focus:ring-offset-gray-900'
                      : 'bg-gradient-to-r from-blue-600 to-cyan-500 hover:from-blue-500 hover:to-cyan-400 text-white shadow-lg hover:shadow-xl focus:ring-blue-500 focus:ring-offset-white'
                    : theme === 'dark'
                      ? 'bg-gray-800 text-gray-500 focus:ring-gray-700 focus:ring-offset-gray-900'
                      : 'bg-gray-100 text-gray-400 focus:ring-gray-300 focus:ring-offset-white'
                )}
                aria-busy={isTransitioning}
              >
                {isTransitioning ? (
                  <>
                    <div className="w-5 h-5 border-2 border-white/30 border-t-white rounded-full animate-spin mr-2" />
                    Setting up your workspace...
                  </>
                ) : (
                  <>
                    Continue to {selectedRoleDetails?.title || 'Onboarding'}
                    <ArrowRight className="ml-2 w-5 h-5 group-hover:translate-x-1 transition-transform" />
                  </>
                )}
              </button>
              
              <p className={cn(
                'mt-4 text-sm max-w-md mx-auto',
                theme === 'dark' ? 'text-gray-500' : 'text-gray-600'
              )}>
                You will be redirected to the specific setup portal for your selected role.
                {!selectedRole && (
                  <span className="block mt-1 text-amber-600 dark:text-amber-400">
                    Please select a role to continue
                  </span>
                )}
              </p>
            </motion.div>

            {/* Security & Help Section */}
            <motion.div 
              initial={{ opacity: 0 }}
              animate={{ opacity: 1 }}
              transition={{ delay: 0.6 }}
              className="text-center space-y-6"
            >
              {/* Security badges */}
              <div className="flex flex-wrap items-center justify-center gap-4">
                {[
                  { icon: Shield, label: 'HIPAA Compliant' },
                  { icon: Lock, label: 'End-to-End Encryption' },
                  { icon: Users, label: 'Role-Based Access' },
                  { icon: Activity, label: 'SOC 2 Type II' }
                ].map((badge, index) => (
                  <motion.div
                    key={badge.label}
                    initial={{ scale: 0, opacity: 0 }}
                    animate={{ scale: 1, opacity: 1 }}
                    transition={{ delay: 0.6 + index * 0.1 }}
                    className={cn(
                      'flex items-center gap-2 px-3 py-2 rounded-lg',
                      theme === 'dark'
                        ? 'bg-gray-800/50 text-gray-300'
                        : 'bg-gray-100 text-gray-700'
                    )}
                  >
                    <badge.icon className="w-4 h-4" />
                    <span className="text-xs font-medium">{badge.label}</span>
                  </motion.div>
                ))}
              </div>

              {/* Help section */}
              <div className="pt-6 border-t border-gray-200 dark:border-gray-800">
                <p className={cn(
                  'text-sm mb-2',
                  theme === 'dark' ? 'text-gray-400' : 'text-gray-600'
                )}>
                  Not sure which role to choose?
                </p>
                <div className="flex items-center justify-center gap-4">
                  <Link
                    to="/help/role-selection"
                    className={cn(
                      'inline-flex items-center gap-2 px-4 py-2 rounded-lg text-sm font-medium',
                      'transition-colors',
                      theme === 'dark'
                        ? 'text-cyan-400 hover:text-cyan-300 hover:bg-gray-800'
                        : 'text-blue-600 hover:text-blue-700 hover:bg-gray-100'
                    )}
                  >
                    <HelpCircle className="w-4 h-4" />
                    View Role Comparison
                  </Link>
                  <Link
                    to="/contact"
                    className={cn(
                      'inline-flex items-center gap-2 px-4 py-2 rounded-lg text-sm font-medium',
                      'transition-colors',
                      theme === 'dark'
                        ? 'text-gray-400 hover:text-white hover:bg-gray-800'
                        : 'text-gray-600 hover:text-gray-900 hover:bg-gray-100'
                    )}
                  >
                    Contact Support
                  </Link>
                </div>
              </div>
            </motion.div>
          </div>
        </main>

        {/* Footer */}
        <motion.footer 
          initial={{ opacity: 0 }}
          animate={{ opacity: 1 }}
          transition={{ delay: 0.8 }}
          className="px-5 py-4 lg:px-8 lg:py-6 relative z-10"
        >
          <div className={cn(
            'text-center text-xs',
            theme === 'dark' ? 'text-gray-500' : 'text-gray-600'
          )}>
            <div className="flex flex-col sm:flex-row items-center justify-center gap-1 sm:gap-4">
              <span>© {new Date().getFullYear()} CustoCare AI. All rights reserved.</span>
              <span className="hidden sm:inline">•</span>
              <div className="flex items-center gap-4">
                <Link
                  to="/privacy"
                  className={cn(
                    'transition-colors hover:underline',
                    theme === 'dark'
                      ? 'text-gray-400 hover:text-gray-300'
                      : 'text-gray-600 hover:text-gray-900'
                  )}
                >
                  Privacy Policy
                </Link>
                <span>•</span>
                <Link
                  to="/terms"
                  className={cn(
                    'transition-colors hover:underline',
                    theme === 'dark'
                      ? 'text-gray-400 hover:text-gray-300'
                      : 'text-gray-600 hover:text-gray-900'
                  )}
                >
                  Terms of Service
                </Link>
                <span>•</span>
                <Link
                  to="/security"
                  className={cn(
                    'transition-colors hover:underline',
                    theme === 'dark'
                      ? 'text-gray-400 hover:text-gray-300'
                      : 'text-gray-600 hover:text-gray-900'
                  )}
                >
                  Security
                </Link>
              </div>
            </div>
          </div>
        </motion.footer>
      </div>

      {/* Right Panel - Testimonial & Benefits */}
      <motion.div 
        initial={{ x: '100%', opacity: 0 }}
        animate={{ x: 0, opacity: 1 }}
        transition={{ type: 'spring', stiffness: 100, damping: 20 }}
        className="hidden lg:flex lg:w-[45%] xl:w-[40%] relative overflow-hidden"
      >
        {/* Background with gradient overlay */}
        <div className="absolute inset-0">
          <img
            src={images.background}
            alt="Modern healthcare facility with doctors using digital tablets"
            className="absolute inset-0 w-full h-full object-cover"
            loading="eager"
            onError={(e) => {
              e.currentTarget.style.display = 'none';
            }}
          />
          <div className={cn(
            'absolute inset-0',
            theme === 'dark'
              ? 'bg-gradient-to-br from-gray-950/95 via-gray-900/90 to-gray-950/95'
              : 'bg-gradient-to-br from-blue-900/40 via-cyan-900/30 to-purple-900/40'
          )} />
        </div>

        {/* Content */}
        <div className="relative z-20 flex flex-col justify-between p-8 xl:p-12 text-white w-full h-full">
          {/* Top Section - Trust Badges */}
          <motion.div 
            initial={{ y: -20, opacity: 0 }}
            animate={{ y: 0, opacity: 1 }}
            transition={{ delay: 0.2 }}
            className="flex items-center justify-between"
          >
            <div className="flex items-center gap-2.5">
              <Shield className="w-5 h-5 text-cyan-300" />
              <span className="text-sm font-medium bg-gradient-to-r from-cyan-300 to-blue-300 bg-clip-text text-transparent">
                Trusted by 500+ Healthcare Facilities
              </span>
            </div>
            <div className="flex items-center gap-2">
              <Activity className="w-4 h-4 text-cyan-300" />
              <span className="text-xs text-cyan-100/80">99.9% Uptime</span>
            </div>
          </motion.div>

          {/* Middle Section - Testimonial & Benefits */}
          <div className="flex-1 flex flex-col justify-center gap-8">
            {/* Testimonial Card */}
            <motion.div 
              initial={{ scale: 0.9, opacity: 0 }}
              animate={{ scale: 1, opacity: 1 }}
              transition={{ delay: 0.4 }}
              className="bg-gradient-to-br from-white/10 to-white/5 backdrop-blur-xl rounded-2xl p-6 xl:p-8 border border-white/10 shadow-2xl"
            >
              <Quote className="w-8 h-8 text-cyan-300 mb-4" />
              <p className="text-lg xl:text-xl italic leading-relaxed mb-6 text-white/95">
                "CustoCare AI has revolutionized how our entire facility operates. The seamless connection between doctors and patients has improved patient outcomes by 40%."
              </p>
              
              <div className="flex items-center gap-4">
                <div className="relative">
                  <div className="w-14 h-14 rounded-full overflow-hidden border-2 border-cyan-400/30">
                    <img
                      src={images.testimonial}
                      alt="Dr. Sarah Jansen, Chief of Medicine"
                      className="w-full h-full object-cover"
                      loading="lazy"
                      onError={(e) => {
                        e.currentTarget.src = images.fallbackAvatar;
                      }}
                    />
                  </div>
                  <div className="absolute -bottom-1 -right-1 w-6 h-6 bg-gradient-to-br from-cyan-400 to-blue-500 rounded-full flex items-center justify-center">
                    <Check className="w-3 h-3 text-white" />
                  </div>
                </div>
                <div>
                  <h4 className="font-bold text-lg bg-gradient-to-r from-white to-cyan-100 bg-clip-text text-transparent">
                    Dr. Sarah Jansen
                  </h4>
                  <p className="text-sm text-cyan-100/90">Chief of Medicine</p>
                  <p className="text-xs text-cyan-100/70 mt-1">St. Mary's Medical Center</p>
                </div>
              </div>
            </motion.div>

            {/* Benefits List */}
            <motion.div 
              initial={{ y: 20, opacity: 0 }}
              animate={{ y: 0, opacity: 1 }}
              transition={{ delay: 0.6 }}
              className="space-y-4"
            >
              <h3 className="text-lg font-semibold text-white/95">Why Healthcare Leaders Choose Us</h3>
              <ul className="space-y-3">
                {[
                  { text: 'Reduced administrative workload by 60%', icon: Sparkles },
                  { text: 'HIPAA & GDPR compliant out of the box', icon: Shield },
                  { text: 'AI-powered clinical decision support', icon: Activity },
                  { text: 'Real-time collaboration across teams', icon: Users }
                ].map((benefit, index) => (
                  <motion.li
                    key={index}
                    initial={{ x: -20, opacity: 0 }}
                    animate={{ x: 0, opacity: 1 }}
                    transition={{ delay: 0.6 + index * 0.1 }}
                    className="flex items-center gap-3"
                  >
                    <div className="w-6 h-6 rounded-full bg-gradient-to-r from-cyan-500/20 to-blue-500/20 flex items-center justify-center">
                      <benefit.icon className="w-3 h-3 text-cyan-300" />
                    </div>
                    <span className="text-sm text-white/90">{benefit.text}</span>
                  </motion.li>
                ))}
              </ul>
            </motion.div>
          </div>

          {/* Bottom Section - Tagline */}
          <motion.div 
            initial={{ y: 20, opacity: 0 }}
            animate={{ y: 0, opacity: 1 }}
            transition={{ delay: 0.8 }}
            className="text-center"
          >
            <div className="inline-flex items-center gap-3 text-cyan-100/90">
              <div className="flex-1 h-px bg-gradient-to-r from-transparent via-cyan-400/50 to-transparent" />
              <span className="text-sm font-medium tracking-wider">CONNECTING CARE, EMPOWERING HEALTH</span>
              <div className="flex-1 h-px bg-gradient-to-l from-transparent via-cyan-400/50 to-transparent" />
            </div>
          </motion.div>
        </div>

        {/* Decorative Elements */}
        <div className="absolute top-0 right-0 w-64 h-64 bg-gradient-to-br from-cyan-500/10 to-blue-500/10 rounded-full blur-3xl" />
        <div className="absolute bottom-0 left-0 w-96 h-96 bg-gradient-to-tr from-purple-500/10 to-pink-500/10 rounded-full blur-3xl" />
      </motion.div>

      {/* Transition Overlay */}
      <AnimatePresence>
        {isTransitioning && (
          <motion.div
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
            className="absolute inset-0 z-50 flex items-center justify-center bg-gray-950/95 backdrop-blur-sm"
          >
            <div className="text-center">
              <motion.div
                animate={{ rotate: 360 }}
                transition={{ duration: 1, repeat: Infinity, ease: "linear" }}
                className="w-16 h-16 border-4 border-cyan-500/30 border-t-cyan-500 rounded-full mx-auto mb-6"
              />
              <h3 className="text-2xl font-bold text-white mb-2">
                Setting up your {selectedRoleDetails?.title.toLowerCase()} workspace
              </h3>
              <p className="text-gray-400 max-w-md mx-auto">
                Configuring permissions, dashboard, and tools specific to your role.
                This will only take a moment...
              </p>
            </div>
          </motion.div>
        )}
      </AnimatePresence>
    </motion.div>
  );
};
